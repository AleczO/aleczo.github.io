Linear Regression in Rust



Simple Linear Regression

Equation Model

$$ y = \alpha + \beta x $$

Cost Function

$$ RSS = \sum_{i=1}^{m} (y_i - \alpha - \beta x_i)^2 $$

Optimal \(\beta\) calculation

$$ \hat{\beta} = \frac{\sum_{i=1}^n(x_i - \bar{x})(y_i - \bar{y}) }{\sum_{i=1}^n (x_i - \bar{x})^2} $$ 

    fn beta_hat( x: &Vec, y: & Vec) -> f64 {
        let (mut num, mut den) = (0.0, 0.0);
        let (x_mean, y_mean) = (mean(x), mean(y));
            
        for i in 0..x.len(){
            num += (x[i] - x_mean) * (y[i] - y_mean);
            den += (x[i] - x_mean) * (x[i] - x_mean);
        }
        
        num / den
    }

Optimal \(\alpha\) calculation

$$ \hat{\alpha} = \bar{y} - \hat{\beta} \bar{x} $$ 

    fn alpha_hat(x: &Vec, y: &Vec, beta: f64) -> f64 {
        mean(y) - beta * mean(x)
    }

Regression Output

$$ \hat{y}_i = \hat{\alpha} + \hat{\beta} x_i $$ 

    fn y_estimation(x: &Vec, y: &Vec) -> Vec {
        let beta: f64 = beta_hat(x, y);
        let alpha : f64 = alpha_hat(x, y, beta);
    
        let mut y_hat: Vec = Vec::new();
    
        for i in 0..x.len(){
            y_hat.push(alpha + beta * x[i]);
        }
    
        y_hat
    }

Correlation Coefficient

$$ r = \frac{ \sum_{i=1}^n(x_i - \bar{x})(y_i - \bar{y}) } { \sqrt{\sum_{i=1}^n(x_i - \bar{x})^2} \sqrt{\sum_{i=1}^n(y_i - \bar{y})^2} } $$

Coefficient of Determination \( r^2 \)

Analysis of Variance (ANOVA) Identity Proof

$$ \sum_{i=1}^n(y_i - \bar{y})^2 = \sum_{i=1}^n(y_i - \hat{y}_i)^2 + \sum_{i=1}^n (\hat{y}_i - \bar{y})^2 $$
$$ \sum_{i=1}^n[(y_i - \hat{y}_i) + (\hat{y}_i - \bar{y})]^2 $$ $$ = \sum_{i=1}^n(y_i - \hat{y}_i)^2 + 2\sum_{i=1}^n (y_i - \hat{y}_i)(\hat{y}_i - \bar{y}) + \sum_{i=1}^n(\hat{y}_i - \bar{y})^2 $$
We need to show that $$ \sum_{i=1}^n (y_i - \hat{y}_i)(\hat{y}_i - \bar{y}) = 0 $$
$$ \hat{y}_i = \hat{\alpha} + \hat{\beta} x_i $$ $$ \hat{\alpha} = \bar{y} - \hat{\beta} \bar{x} $$
$$ \hat{y}_i = \bar{y} - \hat{\beta}\bar{x} + \bar{\beta}x_i $$ $$ \hat{y}_i = \bar{y} + \hat{\beta}(x_i - \bar{x}) $$

$$ = \sum_{i=1}^n (y_i - \hat{y}_i)(\hat{y}_i - \bar{y}) $$ $$ = \sum_{i=1}^n [(y_i - \bar{y}) - \hat{\beta}(x_i - \bar{x})][\hat{\beta}(x_i - \bar{x})] $$ $$ = \hat{\beta} \sum_{i=1}^n (y_i - \bar{y})(x_i - \bar{x}) - \hat{\beta}^2 \sum_{i=1}^n (x_i - x)^2 $$
$$ \hat{\beta} = \frac{\sum_{i=1}^n(x_i - \bar{x})(y_i - \bar{y}) }{\sum_{i=1}^n (x_i - \bar{x})^2} $$ $$ \hat{\beta} {\sum_{i=1}^n (x_i - \bar{x})^2} = \sum_{i=1}^n(x_i - \bar{x})(y_i - \bar{y}) $$
$$ = \hat{\beta}^2 \sum_{i=1}^n (x_i - \bar{x})^2 - \hat{\beta}^2 \sum_{i=1}^n (x_i - x)^2 = 0 $$

\( r^2 \) Derivation

$$ r \frac{\sqrt{\sum_{i=1}^n (y_i - \bar{y})^2}}{\sqrt{\sum_{i=1}^n (x_i - \bar{x})^2}} $$ $$ = \frac{ \sum_{i=1}^n(x_i - \bar{x})(y_i - \bar{y}) } { \sqrt{\sum_{i=1}^n(x_i - \bar{x})^2} \sqrt{\sum_{i=1}^n(y_i - \bar{y})^2}} \frac{\sqrt{\sum_{i=1}^n (y_i - \bar{y})^2}}{\sqrt{\sum_{i=1}^n (x_i - \bar{x})^2}} $$ $$ = \frac{ \sum_{i=1}^n(x_i - \bar{x})(y_i - \bar{y}) } { \sum_{i=1}^n(x_i - \bar{x})^2} = \hat{\beta} $$
$$ r \frac{\sqrt{\sum_{i=1}^n (y_i - \bar{y})^2}}{\sqrt{\sum_{i=1}^n (x_i - \bar{x})^2}} = \hat{\beta} $$ $$ r^2 \frac{\sum_{i=1}^n (y_i - \bar{y})^2}{\sum_{i=1}^n (x_i - \bar{x})^2} = \hat{\beta}^2 $$
From 1.1 $$ \hat{y_i} - \bar{y} = -\hat{\beta}(x_i - \bar{x}) $$ $$ (\hat{y_i} - \bar{y})^2 = \hat{\beta}^2 (x_i - \bar{x})^2 $$ $$ \sum_{i=1}^n (\hat{y_i} - \bar{y})^2 = \hat{\beta}^2\sum_{i=1}^n (x_i - \bar{x})^2 $$ $$ \hat{\beta}^2 = \frac{\sum_{i=1}^n (\hat{y_i} - \bar{y})^2} {\sum_{i=1}^n (x_i - \bar{x})^2} $$
$$ r^2 = \frac{\sum_{i=1}^n (\hat{y_i} - \bar{y})^2} {\sum_{i=1}^n (x_i - \bar{x})^2} \cdot \frac{\sum_{i=1}^n (x_i - \bar{x})^2}{\sum_{i=1}^n (y_i - \bar{y})^2} $$ $$ r^2 = \frac{\sum_{i=1}^n (\hat{y_i} - \bar{y})^2} {\sum_{i=1}^n (y_i - \bar{y})^2} $$

Formula Simplification

$$ r^2 = \frac{\sum_{i=1}^n (\hat{y_i} - \bar{y})^2} {\sum_{i=1}^n (y_i - \bar{y})^2} $$

From 1

$$ \frac{\sum_{i=1}^n (\hat{y_i} - \bar{y})^2} {\sum_{i=1}^n (y_i - \bar{y})^2} = \frac{\sum_{i=1}^n(y_i - \bar{y}_i)^2 - \sum_{i=1}^n ({y}_i - \hat{y})^2}{\sum_{i=1}^n(y_i - \bar{y})^2} $$
$$ r^2 = 1 - \frac{\sum_{i=1}^n ({y}_i - \hat{y})^2}{\sum_{i=1}^n(y_i - \bar{y})^2} $$
$$ r^2 = 1 - \frac{\text{SSE}}{\text{SST}} $$ 

    fn  R2(y: &Vec, y_hat: &Vec) -> f64 {
        let num = RSS(y, y_hat);
        let den = SST(y);
        
        (1.0 - (num / den) as f64)
    }

Error Variance

$$ \sigma^2 = s^2 = \frac{\sum_{i=1}^n(y_i - \hat{y}_i)}{n - 2} $$

Equivalent Formula

from the coefficient of determination

$$ r^2 = 1 - \frac{\sum_{i=1}^n(y_i - \hat{y}_i)^2}{\sum_{i=1}^n(y_i - \bar{y})^2} $$ $$ 1 - r^2 = \frac{\sum_{i=1}^n(y_i - \hat{y}_i)^2}{\sum_{i=1}^n(y_i - \bar{y})^2} $$ $$ (1 - r^2)\sum_{i=1}^n(y_i - \bar{y})^2 = \sum_{i=1}^n(y_i - \hat{y}_i)^2 $$ $$ \frac{(1 - r^2)\sum_{i=1}^n(y_i - \bar{y})^2}{n - 2} = \frac{\sum_{i=1}^n(y_i - \hat{y}_i)^2}{n - 2} = s^2 $$
$$ s^2 = \frac{(1-r^2) \sum_{i=1}^n(y_i - \bar{y})^2}{n-2} $$

Estimate of the Error Standard Deviation

$$ s = \sqrt{\frac{\sum_{i=1}^n(y_i - \hat{y}_i)^2}{n - 2}} = \sqrt{\frac{\text{SSE}}{n - 2}} $$ 

    fn SE(y: &Vec, y_hat: &Vec) -> f64 {
        (SSE(y, &y_hat) / ((y.len() - 2) as f64)).sqrt()
    }

Standard Deviation of \( \hat{\alpha}\) and \( \hat{\beta} \)

$$ s_{\hat{\alpha}} = s \sqrt{\frac{1}{n} + \frac{\bar{x}^2}{\sum_{i=1}^n (x_i - \bar{x})^2} } $$
$$ s_{\hat{\beta}} = \frac{s}{\sqrt{\sum_{i=1}^n(x_i - \bar{x})^2}} $$

Mean 


        fn mean(vec: &Vec ) -> f64 {
            vec.iter().sum::() / (vec.len() as f64)
        }

RSS (Sum of Squared Errors)

$$ \text{SSE} = \text{RSS} = \sum_{i=1}^n (y_i - \hat{y}_i)^2 $$ 
 
        fn RSS(y: &Vec, y_hat: &Vec) -> f64 {
            let mut sum: f64 = 0.0;
                
            for i in 0..y.len(){
                sum += (y[i] - y_hat[i]) * (y[i] - y_hat[i]);
            }
                
            sum
        }

SST (Total Sum of Squares)

$$ SST = \sum_{i=1}^n (y_i - \bar{y})^2 $$ 
 
        fn SST(y: &Vec) -> f64 {
            let mut sum: f64 = 0.0;
            
            for i in 0..y.len(){
                sum += (y[i] - mean(y)) * (y[i] - mean(y));
            }
        
            sum
        }

SSR (Regression Sum of Squares)

$$ SSR = \sum_{i=1}^n (\hat{y}_i - \bar{y})^2 $$ 
 
        fn SSR(y: &Vec, y_hat: &Vec) -> f64 {
            let mut sum = 0.0;
            
            for i in 0..y.len(){
                sum += (y_hat[i] - mean(y)) * (y_hat[i] - mean(y));
            }
            
            sum
        }
