Linear Regression in Rust



Simple Linear Regression

Equation Model

$$ y = \alpha + \beta x $$

Cost Function

$$ RSS = \sum_{i=1}^{m} (y_i - \alpha - \beta x_i)^2 $$

Optimal \(\beta\) calculation

$$ \hat{\beta} = \frac{\sum_{i=1}^n(x_i - \bar{x})(y_i - \bar{y}) }{\sum_{i=1}^n (x_i - \bar{x})^2} $$ 

    fn beta_hat( x: &Vec, y: & Vec) -> f64 {
        let (mut nom, mut den) = (0.0, 0.0);
        
        let (x_mean, y_mean) = (mean(x), mean(y));
        
        
        for i in 0..x.len(){
            nom += (x[i] - x_mean) * (y[i] - y_mean);
            den += (x[i] - x_mean) * (x[i] - x_mean);
        }
        
        nom / den
    }

Optimal \(\alpha\) calculation

$$ \hat{\alpha} = \bar{y} - \hat{\beta} \bar{x} $$ 

    fn alpha_hat(x: &Vec, y: &Vec, beta: f64) -> f64 {
        mean(y) - beta * mean(x)
    }

Regression Output

$$ \hat{y}_i = \hat{\alpha} + \hat{\beta} x_i $$ 

    fn y_estimation(x: &Vec, y: &Vec) -> Vec {
        let beta: f64 = beta_hat(x, y);
        let alpha : f64 = alpha_hat(x, y, beta);
    
        let mut y_hat: Vec = Vec::new();
    
        for i in 0..x.len(){
            y_hat.push(alpha + beta * x[i]);
        }
    
        y_hat
    }

Mean 


        fn mean(vec: &Vec ) -> f64 {
            vec.iter().sum::() / (vec.len() as f64)
        }

RSS

$$ RSS = \sum_{i=1}^n (y_i - \hat{y}_i)^2 $$ 

        fn RSS(y: &Vec, y_hat: &Vec) -> f64 {
            let mut sum: f64 = 0.0;
                
            for i in 0..y.len(){
                sum += (y[i] - y_hat[i]) * (y[i] - y_hat[i]);
            }
                
            sum
        }
